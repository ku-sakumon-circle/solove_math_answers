\begin{prob}{017}{}{無理数の証明}
 次に挙げるものが無理数であることを証明せよ。
 \begin{enumerate}
  \item $\sqrt{2}$ \hosi 1
  \item $\log_2 10$  \hosi 1
  \item $3^{\frac{1}{3}}$ \hosi 1
  \item $\tan 1^\circ$ \hosi 4 (京大)
  \item $\cos 1^\circ$ \hosi 6
  \item $3^{\frac{1}{3}}p+\sqrt{2}q$ (ただし$p, q$は正の有理数) \hosi 4 (阪大 類)
 \end{enumerate}
\end{prob}
