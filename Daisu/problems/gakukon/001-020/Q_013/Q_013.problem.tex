\begin{prob}{013}{\hosi 1}{Daisu gakukon テンプレート問題}
 $p=2^{10} - 3$とおき, 数列$a_n, b_n$を以下の式で定める.
\begin{align*}
&a_0=0,\quad a_1 = 1,\quad a_{n+2} = 2a_{n+1} +2a_n \quad (n=0,1,\dots) \\
&b_0=0, \quad b_1 = 1,\quad b_{n+2} = 2b_{n+1} +(p+2)b_n \quad (n=0,1,\dots)
\end{align*}
\begin{enumerate}
    \item $a_n,b_n$をそれぞれ$n$で表せ.
    \item $a_{1024}$を$p$で割った余りを求めよ. ただし, 整数$m$に対して$m^p\equiv m\pmod{p}$であることを用いてもよい.
\end{enumerate}
\end{prob}
